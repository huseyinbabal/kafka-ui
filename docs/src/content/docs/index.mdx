import { Card, CardGrid } from '@astrojs/starlight/components'

<CardGrid>
  <Card title="Live topology" icon="random">
    Brokers, topics and consumer groups laid out as a flow. Line width and motion follow msg/s; consumers joining or
    leaving are highlighted as it happens.
  </Card>
  <Card title="Message browser" icon="magnifier">
    Seek by offset or time, tail live, filter with regex on the server, and decode Avro, Protobuf and JSON Schema
    through Schema Registry.
  </Card>
  <Card title="Consumer lag" icon="warning">
    Per-partition lag, lag trends, and a "catch up in" estimate for every group. Offset resets with a mandatory
    preview.
  </Card>
  <Card title="Analytics" icon="seti:graphql">
    Throughput, top topics, partition skew and hygiene checks, sampled in the background and kept for 7 days.
  </Card>
  <Card title="Safe by default" icon="approve-check">
    `admin` and `viewer` roles, read-only clusters, protected internal topics, typed confirmations and an audit log.
  </Card>
  <Card title="Any Kafka" icon="puzzle">
    Talks only the Kafka protocol (3.x+, KRaft or ZooKeeper). Single Go binary with the UI embedded and SQLite for
    state.
  </Card>
</CardGrid>
